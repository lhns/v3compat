# --- Exec: the client must follow every way into a new program, in every libc shape. ---
set(B ${CMAKE_CURRENT_BINARY_DIR})
function(shape name src)  # a glibc build of src: dyn, static, spie
  foreach(kind dyn static spie)
    add_executable(${name}_${kind} ${src})
    target_compile_options(${name}_${kind} PRIVATE ${ARGN})
    target_link_libraries(${name}_${kind} PRIVATE ${CMAKE_DL_LIBS} pthread)
  endforeach()
  target_link_options(${name}_static PRIVATE -static)
  target_compile_options(${name}_spie PRIVATE -fPIE)
  target_link_options(${name}_spie PRIVATE -static-pie)
  target_compile_options(${name}_dyn PRIVATE -fPIE)
endfunction()
shape(lz lzcnt.c -mlzcnt)
shape(execer execer.c)
shape(selfprobe selfprobe.c)
shape(clone3 clone3.c)

# musl is a different libc and loader; CI installs musl-tools.

set(MUSL_OUT "")
function(musl name src)
  add_custom_command(OUTPUT ${M}/${name} COMMAND ${CMAKE_COMMAND} -E make_directory ${M}
    COMMAND ${MUSL_GCC} -O1 -g -w ${ARGN} -o ${M}/${name} ${CMAKE_CURRENT_SOURCE_DIR}/${src}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/${src} VERBATIM)
  set(MUSL_OUT ${MUSL_OUT} ${M}/${name} PARENT_SCOPE)
endfunction()
if(MUSL_GCC)
  musl(lz_mdyn lzcnt.c -mlzcnt)
  musl(lz_mstatic lzcnt.c -mlzcnt -static)
  musl(execer_mdyn execer.c)
  musl(execer_mstatic execer.c -static)
  musl(selfprobe_mdyn selfprobe.c)
  musl(selfprobe_mstatic selfprobe.c -static)
  musl(clone3_mdyn clone3.c)
  musl(clone3_mstatic clone3.c -static)
  add_custom_target(musl_progs ALL DEPENDS ${MUSL_OUT})
endif()
